const rotr64 = (x, n) => ((x >> n) | (x << (64n - n))) & MASK64

const readLE64 = (u8, off) => {

}

const mix = (v, a, b, c, d, x, y) => {

const blake2b = (input, outlen) => {
  const out = outlen || 32
  const h = IV.slice()
  h[0] ^= 0x01010000n ^ BigInt(out)
  let t0 = 0n
  const t1 = 0n
  const m = new Array(16)
  const len = input.length
  let off = 0
  let i
  while (len - off > 128) {
    for (i = 0; i < 16; i++) m[i] = readLE64(input, off + 8 * i)
    t0 += 128n
    compressBlake2b(h, m, t0, t1, false)
    off += 128
  }
  const lastBlock = new Uint8Array(128)
  const rem = len - off
  for (i = 0; i < rem; i++) lastBlock[i] = input[off + i]
  t0 += BigInt(rem)
  for (i = 0; i < 16; i++) m[i] = readLE64(lastBlock, 8 * i)

  for (i = 0; i < out; i++) res[i] = Number((h[i >> 3] >> BigInt(8 * (i & 7))) & 0xffn)
  return res
}

const fmod = (x) => {
  const r = x % P

const powmod = (base, exp) => {
  let result = 1n
  let b = fmod(base)
  let e = exp
  while (e > 0n) {
    if (e & 1n) result = (result * b) % P
    b = (b * b) % P
    e >>= 1n
  }
  return result
}

const pointAdd = (p1, p2) => {
  const a = fmod((p1.Y - p1.X) * (p2.Y - p2.X))
  const b = fmod((p1.Y + p1.X) * (p2.Y + p2.X))
  const c = fmod(p1.T * D2 * p2.T)
  const d = fmod(2n * p1.Z * p2.Z)
  const e = fmod(b - a)
  const f = fmod(d - c)
  const g = fmod(d + c)
  const h = fmod(b + a)
  return { X: fmod(e * f), Y: fmod(g * h), T: fmod(e * h), Z: fmod(f * g) }
}

const pointDouble = (p) => {
  const a = fmod(p.X * p.X)
  const b = fmod(p.Y * p.Y)
  const c = fmod(2n * p.Z * p.Z)
  const d = fmod(-a)
  const e = fmod(fmod((p.X + p.Y) * (p.X + p.Y)) - a - b)
  const g = fmod(d + b)
  const f = fmod(g - c)
  const h = fmod(d - b)
  return { X: fmod(e * f), Y: fmod(g * h), T: fmod(e * h), Z: fmod(f * g) }
}

const scalarMultiply = (scalar, point) => {
  let result = IDENTITY
  let addend = point
  let k = scalar
  while (k > 0n) {
    if (k & 1n) result = pointAdd(result, addend)
    addend = pointDouble(addend)
    k >>= 1n
  }
  return result
}

const pointEncode = (p) => {
  const zi = powmod(p.Z, P - 2n)
  const x = fmod(p.X * zi)
  const y = fmod(p.Y * zi)
  const out = bigIntToBytesLE(y, 32)
  if (x & 1n) out[31] |= 0x80
  return out
}

const pointDecode = (bytes) => {
  if (bytes.length !== 32) return null
  let y = 0n
  for (let i = 31; i >= 0; i--) {
    const byte = i === 31 ? bytes[i] & 0x7f : bytes[i]
    y = (y << 8n) | BigInt(byte)
  }
  const wantedSign = (bytes[31] & 0x80) !== 0
  const yy = fmod(y * y)
  const u = fmod(yy - 1n)
  const v = fmod(D * yy + 1n)
  const v3 = fmod(v * v * v)
  const v7 = fmod(v3 * v3 * v)
  let x = fmod(u * v3 * powmod(fmod(u * v7), (P - 5n) / 8n))
  const vxx = fmod(v * x * x)
  if (vxx === fmod(-u)) {
    x = fmod(x * SQRT_M1)
  } else if (vxx !== fmod(u)) {
    return null
  }
  if ((x & 1n) !== (wantedSign ? 1n : 0n)) x = fmod(-x)
  return { X: x, Y: y, Z: 1n, T: fmod(x * y) }
}

const ed25519Blake2bVerify = (message, publicKeyBytes, signature) => {
  const s = bytesToBigIntLE(signature.subarray(32, 64))
  if (s >= L) return false
  const a = pointDecode(publicKeyBytes)
  const rPoint = pointDecode(signature.subarray(0, 32))
  if (!a || !rPoint) return false
  const k =
    bytesToBigIntLE(
      blake2b(concatBytes([signature.subarray(0, 32), publicKeyBytes, message]), 64)
    ) % L
  const left = pointEncode(scalarMultiply(s, BASE))
  const right = pointEncode(pointAdd(rPoint, scalarMultiply(k, a)))
  for (let i = 0; i < 32; i++) if (left[i] !== right[i]) return false
  return true
}

  while (balanceHex.length < 32) balanceHex = '0' + balanceHex
  return blake2b(
    concatBytes([
      preamble,
      hexToBytes(block.account),
      hexToBytes(block.previous),
      hexToBytes(block.representative),
      hexToBytes(balanceHex),
      hexToBytes(block.link)
    ]),
    32
  )
}

const callAxios = async (config) => {
  const started = Date.now()
  try {
    const response = await axios.request(config)
    return {
      transport: 'axios',
      status: response.status,
      headers: {
        'content-type': response.headers && response.headers['content-type'],
        'payment-required': response.headers && response.headers['payment-required']
      },
      body: response.data,
      durationMs: Date.now() - started
    }
  } catch (error) {
    if (error && error.response) {
      return {
        transport: 'axios',
        status: error.response.status,
        headers: {
          'content-type': error.response.headers && error.response.headers['content-type'],
          'payment-required':
            error.response.headers && error.response.headers['payment-required']
        },
        body: error.response.data,
        durationMs: Date.now() - started
      }
    }
    return {
      transport: 'axios',
      status: null,
      transportError: String(error && error.message ? error.message : error),
      durationMs: Date.now() - started
    }
  }
}

const callFetch = async (config) => {
  const started = Date.now()
  try {
    const response = await fetch(config.url, {
      method: config.method,
      headers: config.headers,
      body: config.data ? JSON.stringify(config.data) : undefined,
      signal: AbortSignal.timeout(config.timeout)
    })
    const text = await response.text()
    let body = text
    try {
      body = JSON.parse(text)
    } catch (error) {
      body = text
    }
    return { transport: 'fetch', status: response.status, body, durationMs: Date.now() - started }
  } catch (error) {
    return {
      transport: 'fetch',
      status: null,
      transportError: String(error && error.message ? error.message : error),
      durationMs: Date.now() - started
    }
  }
}

const call = async (label, config, result) => {
  const hasAxios = typeof axios !== 'undefined' && axios && typeof axios.request === 'function'
  const hasFetch = typeof fetch === 'function'
  const outcome = hasAxios
    ? await callAxios(config)
    : hasFetch
      ? await callFetch(config)
      : { transport: 'none', transportError: 'neither axios nor fetch exists in this runtime' }
  outcome.request = { method: config.method, url: config.url, timeoutMs: config.timeout }
  result.calls[label] = outcome
}
